########## src/algorithms/M2Search.mod ##########

IMPLEMENTATION MODULE M2Search;

PROCEDURE Linear(a:ARRAY OF CARDINAL;

key:CARDINAL):INTEGER;

VAR i:CARDINAL;

BEGIN FOR i:=0 TO HIGH(a)DO IF a[i]=key THEN RETURN VAL(INTEGER,i)END END;

RETURN -1 END Linear;

PROCEDURE LowerBound(a:ARRAY OF CARDINAL;

key:CARDINAL):CARDINAL;

VAR lo,hi,mid:CARDINAL;

BEGIN lo:=0;

hi:=HIGH(a)+1;

WHILE lo<hi DO mid:=lo+(hi-lo)DIV 2;

IF a[mid]<key THEN lo:=mid+1 ELSE hi:=mid END END;

RETURN lo END LowerBound;

PROCEDURE UpperBound(a:ARRAY OF CARDINAL;

key:CARDINAL):CARDINAL;

VAR lo,hi,mid:CARDINAL;

BEGIN lo:=0;

hi:=HIGH(a)+1;

WHILE lo<hi DO mid:=lo+(hi-lo)DIV 2;

IF a[mid]<=key THEN lo:=mid+1 ELSE hi:=mid END END;

RETURN lo END UpperBound;

PROCEDURE Binary(a:ARRAY OF CARDINAL;

key:CARDINAL):INTEGER;

VAR p:CARDINAL;

BEGIN p:=LowerBound(a,key);

IF(p<=HIGH(a))AND(a[p]=key)THEN RETURN VAL(INTEGER,p)END;

RETURN -1 END Binary;

END M2Search.

########## src/algorithms/M2Sort.mod ##########

IMPLEMENTATION MODULE M2Sort;

PROCEDURE Swap(VAR a,b:CARDINAL);

VAR t:CARDINAL;

BEGIN t:=a;

a:=b;

b:=t END Swap;

PROCEDURE InsertionSort(VAR a:ARRAY OF CARDINAL);

VAR i,j,x:CARDINAL;

BEGIN IF HIGH(a)<1 THEN RETURN END;

FOR i:=1 TO HIGH(a)DO x:=a[i];

j:=i;

WHILE(j>0)AND(a[j-1]>x)DO a[j]:=a[j-1];

DEC(j)END;

a[j]:=x END END InsertionSort;

PROCEDURE SelectionSort(VAR a:ARRAY OF CARDINAL);

VAR i,j,m:CARDINAL;

BEGIN IF HIGH(a)<1 THEN RETURN END;

FOR i:=0 TO HIGH(a)-1 DO m:=i;

FOR j:=i+1 TO HIGH(a)DO IF a[j]<a[m]THEN m:=j END END;

IF m#i THEN Swap(a[i],a[m])END END END SelectionSort;

PROCEDURE Sift(VAR a:ARRAY OF CARDINAL;

root,last:CARDINAL);

VAR child,t:CARDINAL;

BEGIN WHILE root*2+1<=last DO child:=root*2+1;

IF(child<last)AND(a[child]<a[child+1])THEN INC(child)END;

IF a[root]>=a[child]THEN RETURN END;

t:=a[root];

a[root]:=a[child];

a[child]:=t;

root:=child END END Sift;

PROCEDURE HeapSort(VAR a:ARRAY OF CARDINAL);

VAR start,last,t:CARDINAL;

BEGIN IF HIGH(a)<1 THEN RETURN END;

start:=(HIGH(a)+1)DIV 2;

WHILE start>0 DO DEC(start);

Sift(a,start,HIGH(a))END;

last:=HIGH(a);

WHILE last>0 DO t:=a[0];

a[0]:=a[last];

a[last]:=t;

DEC(last);

Sift(a,0,last)END END HeapSort;

END M2Sort.

PROCEDURE FNV1a32(s:ARRAY OF CHAR):CARDINAL;

VAR h,i:CARDINAL;

BEGIN h:=2166136261;

i:=0;

WHILE(i<=HIGH(s))AND(s[i]#0C)DO h:=(h XOR ORD(s[i]))*16777619;

INC(i)END;

RETURN h END FNV1a32;

PROCEDURE DJB2(s:ARRAY OF CHAR):CARDINAL;

VAR h,i:CARDINAL;

BEGIN h:=5381;

i:=0;

WHILE(i<=HIGH(s))AND(s[i]#0C)DO h:=h*33+ORD(s[i]);

INC(i)END;

RETURN h END DJB2;

PROCEDURE Mix32(x:CARDINAL):CARDINAL;

BEGIN x:=x XOR(x DIV 65536);

x:=x*16#7FEB352D;

x:=x XOR(x DIV 32768);

x:=x*16#846CA68B;

RETURN x XOR(x DIV 65536)END Mix32;

END M2Hash.

IMPLEMENTATION MODULE M2Heap;

PROCEDURE Sift(VAR a:ARRAY OF CARDINAL;

r,last:CARDINAL);

VAR c,t:CARDINAL;

BEGIN WHILE r*2+1<=last DO c:=r*2+1;

IF(c<last)AND(a[c]<a[c+1])THEN INC(c)END;

IF a[r]>=a[c]THEN RETURN END;

t:=a[r];

a[r]:=a[c];

a[c]:=t;

r:=c END END Sift;

PROCEDURE HeapSort(VAR a:ARRAY OF CARDINAL);

VAR s,l,t:CARDINAL;

BEGIN IF HIGH(a)<1 THEN RETURN END;

s:=(HIGH(a)+1)DIV 2;

WHILE s>0 DO DEC(s);

Sift(a,s,HIGH(a))END;

l:=HIGH(a);

WHILE l>0 DO t:=a[0];

a[0]:=a[l];

a[l]:=t;

DEC(l);

Sift(a,0,l)END END HeapSort;

PROCEDURE IsHeap(a:ARRAY OF CARDINAL):BOOLEAN;

VAR i:CARDINAL;

BEGIN IF HIGH(a)<1 THEN RETURN TRUE END;

FOR i:=1 TO HIGH(a)DO IF a[(i-1)DIV 2]<a[i]THEN RETURN FALSE END END;

RETURN TRUE END IsHeap;

END M2Heap.

IMPLEMENTATION MODULE M2Queue;

FROM M2Errors IMPORT ReportError,ErrOutOfMemory,ErrUnderflow,ErrOverflow;

PROCEDURE Create(capacity:CARDINAL):Queue;

VAR q:Queue;

BEGIN IF capacity=0 THEN capacity:=16 END;

NEW(q);

IF q=NIL THEN ReportError(ErrOutOfMemory,"Queue");

RETURN NIL END;

NEW(q^.data,capacity);

IF q^.data=NIL THEN DISPOSE(q);

ReportError(ErrOutOfMemory,"Queue data");

RETURN NIL END;

q^.cap:=capacity;

q^.head:=0;

q^.len:=0;

RETURN q END Create;

PROCEDURE Destroy(VAR q:Queue);

BEGIN IF q#NIL THEN IF q^.data#NIL THEN DISPOSE(q^.data)END;

DISPOSE(q);

q:=NIL END END Destroy;

PROCEDURE Enqueue(q:Queue;

value:CARDINAL):BOOLEAN;

VAR t:CARDINAL;

BEGIN IF q=NIL THEN RETURN FALSE END;

IF q^.len=q^.cap THEN ReportError(ErrOverflow,"Queue full");

RETURN FALSE END;

t:=(q^.head+q^.len)MOD q^.cap;

q^.data^[t]:=value;

INC(q^.len);

RETURN TRUE END Enqueue;

PROCEDURE Dequeue(q:Queue;

VAR value:CARDINAL):BOOLEAN;

BEGIN IF(q=NIL)OR(q^.len=0)THEN ReportError(ErrUnderflow,"Queue.Dequeue");

RETURN FALSE END;

value:=q^.data^[q^.head];

q^.head:=(q^.head+1)MOD q^.cap;

DEC(q^.len);

RETURN TRUE END Dequeue;

PROCEDURE Front(q:Queue;

VAR value:CARDINAL):BOOLEAN;

BEGIN IF(q=NIL)OR(q^.len=0)THEN RETURN FALSE END;

value:=q^.data^[q^.head];

RETURN TRUE END Front;

PROCEDURE Size(q:Queue):CARDINAL;

BEGIN IF q=NIL THEN RETURN 0 ELSE RETURN q^.len END END Size;

PROCEDURE IsEmpty(q:Queue):BOOLEAN;

BEGIN RETURN(q=NIL)OR(q^.len=0)END IsEmpty;

PROCEDURE Clear(q:Queue);

BEGIN IF q#NIL THEN q^.head:=0;

q^.len:=0 END END Clear;

END M2Queue.

########## src/collections/M2Vector.mod ##########

IMPLEMENTATION MODULE M2Vector;

FROM Storage IMPORT ALLOCATE,DEALLOCATE;

FROM SYSTEM IMPORT TSIZE,ADDRESS;

FROM M2Errors IMPORT ReportError,ErrOutOfMemory,ErrIndexOutOfBounds,ErrUnderflow,ErrOverflow;

PROCEDURE Slot(data:ADDRESS;

idx:CARDINAL):POINTER TO CARDINAL;

VAR p:POINTER TO CARDINAL;

BEGIN p:=CAST(POINTER TO CARDINAL,data+idx*TSIZE(CARDINAL));

RETURN p END Slot;

PROCEDURE Create(initCap:CARDINAL):Vector;

VAR v:Vector;

BEGIN IF initCap=0 THEN initCap:=8 END;

NEW(v);

IF v=NIL THEN ReportError(ErrOutOfMemory,"Vector descriptor");

RETURN NIL END;

ALLOCATE(v^.data,initCap*TSIZE(CARDINAL));

IF v^.data=NIL THEN DISPOSE(v);

ReportError(ErrOutOfMemory,"Vector data");

RETURN NIL END;

v^.len:=0;

v^.cap:=initCap;

RETURN v END Create;

PROCEDURE Destroy(VAR v:Vector);

BEGIN IF v#NIL THEN IF v^.data#NIL THEN DEALLOCATE(v^.data,v^.cap*TSIZE(CARDINAL))END;

DISPOSE(v);

v:=NIL END END Destroy;

PROCEDURE Reserve(v:Vector;

newCap:CARDINAL):BOOLEAN;

VAR nd:ADDRESS;

i:CARDINAL;

p,q:POINTER TO CARDINAL;

BEGIN IF v=NIL THEN RETURN FALSE END;

IF newCap<=v^.cap THEN RETURN TRUE END;

IF newCap>MAX(CARDINAL)DIV TSIZE(CARDINAL)THEN ReportError(ErrOverflow,"Vector.Reserve");

RETURN FALSE END;

ALLOCATE(nd,newCap*TSIZE(CARDINAL));

IF nd=NIL THEN ReportError(ErrOutOfMemory,"Vector.Reserve");

RETURN FALSE END;

i:=0;

WHILE i<v^.len DO p:=Slot(v^.data,i);

q:=Slot(nd,i);

q^:=p^;

INC(i)END;

IF v^.data#NIL THEN DEALLOCATE(v^.data,v^.cap*TSIZE(CARDINAL))END;

v^.data:=nd;

v^.cap:=newCap;

RETURN TRUE END Reserve;

PROCEDURE Push(v:Vector;

val:CARDINAL):BOOLEAN;

VAR p:POINTER TO CARDINAL;

n:CARDINAL;

BEGIN IF v=NIL THEN RETURN FALSE END;

IF v^.len=v^.cap THEN IF v^.cap>MAX(CARDINAL)DIV 2 THEN ReportError(ErrOverflow,"Vector.Push");

RETURN FALSE END;

n:=v^.cap*2;

IF n=0 THEN n:=8 END;

IF NOT Reserve(v,n)THEN RETURN FALSE END END;

p:=Slot(v^.data,v^.len);

p^:=val;

INC(v^.len);

RETURN TRUE END Push;

PROCEDURE Pop(v:Vector;

VAR val:CARDINAL):BOOLEAN;

VAR p:POINTER TO CARDINAL;

BEGIN IF(v=NIL)OR(v^.len=0)THEN ReportError(ErrUnderflow,"Vector.Pop");

RETURN FALSE END;

DEC(v^.len);

p:=Slot(v^.data,v^.len);

val:=p^;

RETURN TRUE END Pop;

PROCEDURE Get(v:Vector;

idx:CARDINAL;

VAR val:CARDINAL):BOOLEAN;

VAR p:POINTER TO CARDINAL;

BEGIN IF(v=NIL)OR(idx>=v^.len)THEN ReportError(ErrIndexOutOfBounds,"Vector.Get");

RETURN FALSE END;

p:=Slot(v^.data,idx);

val:=p^;

RETURN TRUE END Get;

PROCEDURE Set(v:Vector;

idx,val:CARDINAL):BOOLEAN;

VAR p:POINTER TO CARDINAL;

BEGIN IF(v=NIL)OR(idx>=v^.len)THEN ReportError(ErrIndexOutOfBounds,"Vector.Set");

RETURN FALSE END;

p:=Slot(v^.data,idx);

p^:=val;

RETURN TRUE END Set;

PROCEDURE Size(v:Vector):CARDINAL;

BEGIN IF v=NIL THEN RETURN 0 ELSE RETURN v^.len END END Size;

PROCEDURE Capacity(v:Vector):CARDINAL;

BEGIN IF v=NIL THEN RETURN 0 ELSE RETURN v^.cap END END Capacity;

PROCEDURE Clear(v:Vector);

BEGIN IF v#NIL THEN v^.len:=0 END END Clear;

PROCEDURE IsEmpty(v:Vector):BOOLEAN;

BEGIN RETURN(v=NIL)OR(v^.len=0)END IsEmpty;

END M2Vector.

PROCEDURE IsPowerOfTwo(n:CARDINAL):BOOLEAN;

BEGIN IF n=0 THEN RETURN FALSE END;

RETURN (BITSET(n)*BITSET(n-1))=BITSET{} END IsPowerOfTwo;

PROCEDURE Log2Floor(n:CARDINAL):CARDINAL;

VAR r:CARDINAL;

BEGIN IF n=0 THEN RETURN 0 END;

r:=0;

WHILE n>1 DO n:=n DIV 2;

INC(r) END;

RETURN r END Log2Floor;

PROCEDURE NextPowerOfTwo(n:CARDINAL):CARDINAL;

VAR p:CARDINAL;

BEGIN IF n<=1 THEN RETURN 1 END;

IF IsPowerOfTwo(n) THEN RETURN n END;

p:=1;

WHILE p<n DO IF p>MAX(CARDINAL) DIV 2 THEN RETURN n END;

p:=p*2 END;

RETURN p END NextPowerOfTwo;

END M2Limits.

IMPLEMENTATION MODULE M2Path;

FROM M2Strings IMPORT Length,Copy,Append,SubString;

PROCEDURE IsAbsolute(path:ARRAY OF CHAR):BOOLEAN;

BEGIN RETURN(Length(path)>0)AND(path[0]='/')END IsAbsolute;

PROCEDURE Join(a,b:ARRAY OF CHAR;

VAR out:ARRAY OF CHAR):BOOLEAN;

VAR n:CARDINAL;

BEGIN Copy(a,out);

n:=Length(out);

IF(n>0)AND(out[n-1]#'/')AND(Length(b)>0)AND(b[0]#'/')THEN Append("/",out)END;

IF IsAbsolute(b)THEN Copy(b,out)ELSE Append(b,out)END;

RETURN Length(out)>0 END Join;

PROCEDURE Basename(path:ARRAY OF CHAR;

VAR out:ARRAY OF CHAR);

VAR i,n:CARDINAL;

BEGIN n:=Length(path);

i:=n;

WHILE i>0 DO DEC(i);

IF path[i]='/'THEN INC(i);

EXIT END END;

SubString(path,i,n-i,out)END Basename;

PROCEDURE Dirname(path:ARRAY OF CHAR;

VAR out:ARRAY OF CHAR);

VAR i,n:CARDINAL;

BEGIN n:=Length(path);

i:=n;

WHILE i>0 DO DEC(i);

IF path[i]='/'THEN IF i=0 THEN SubString(path,0,1,out)ELSE SubString(path,0,i,out)END;

RETURN END END;

Copy(".",out)END Dirname;

PROCEDURE Extension(path:ARRAY OF CHAR;

VAR out:ARRAY OF CHAR);

VAR i,n,b:CARDINAL;

BEGIN n:=Length(path);

b:=0;

FOR i:=0 TO n-1 DO IF path[i]='/'THEN b:=i+1 END END;

i:=n;

WHILE i>b DO DEC(i);

IF path[i]='.'THEN SubString(path,i,n-i,out);

RETURN END END;

Copy("",out)END Extension;

END M2Path.

PROCEDURE CountFields(line:ARRAY OF CHAR):CARDINAL;

VAR i,n:CARDINAL;

q:BOOLEAN;

BEGIN n:=1;

i:=0;

q:=FALSE;

WHILE(i<=HIGH(line))AND(line[i]#0C)DO IF line[i]='"'THEN IF q AND(i<HIGH(line))AND(line[i+1]='"')THEN INC(i)ELSE q:=NOT q END ELSIF(line[i]=',')AND NOT q THEN INC(n)END;

INC(i)END;

RETURN n END CountFields;

PROCEDURE ParseRow(line:ARRAY OF CHAR;

VAR fields:ARRAY OF ARRAY OF CHAR;

VAR count:CARDINAL):BOOLEAN;

VAR i,j:CARDINAL;

q:BOOLEAN;

BEGIN count:=0;

i:=0;

j:=0;

q:=FALSE;

WHILE(i<=HIGH(line))AND(line[i]#0C)DO IF line[i]='"'THEN IF q AND(i<HIGH(line))AND(line[i+1]='"')THEN IF j>=HIGH(fields[count])THEN RETURN FALSE END;

fields[count][j]:='"';

INC(j);

INC(i)ELSE q:=NOT q END ELSIF(line[i]=',')AND NOT q THEN fields[count][j]:=0C;

IF count>=HIGH(fields)THEN RETURN FALSE END;

INC(count);

j:=0 ELSE IF j>=HIGH(fields[count])THEN RETURN FALSE END;

fields[count][j]:=line[i];

INC(j)END;

INC(i)END;

IF q THEN RETURN FALSE END;

fields[count][j]:=0C;

INC(count);

RETURN TRUE END ParseRow;

END M2CSV.

PROCEDURE FindValue(line,key:ARRAY OF CHAR;

VAR value:ARRAY OF CHAR):BOOLEAN;

VAR i,j,k,n:CARDINAL;

BEGIN n:=Length(line);

i:=0;

WHILE(i<n)AND((line[i]=' ')OR(line[i]=9C))DO INC(i)END;

j:=0;

WHILE(j<Length(key))AND(i<n)DO IF line[i]#key[j]THEN RETURN FALSE END;

INC(i);

INC(j)END;

IF j#Length(key)THEN RETURN FALSE END;

WHILE(i<n)AND(line[i]=' ')DO INC(i)END;

IF(i>=n)OR(line[i]#'=')THEN RETURN FALSE END;

INC(i);

WHILE(i<n)AND(line[i]=' ')DO INC(i)END;

k:=0;

WHILE(i<n)AND(line[i]#'#')AND(line[i]#';')AND(k<HIGH(value))DO value[k]:=line[i];

INC(k);

INC(i)END;

value[k]:=0C;

RETURN TRUE END FindValue;

VAR value:BOOLEAN):BOOLEAN;

BEGIN IF Equal(s,"true")OR Equal(s,"yes")OR Equal(s,"1")THEN value:=TRUE;

RETURN TRUE END;

IF Equal(s,"false")OR Equal(s,"no")OR Equal(s,"0")THEN value:=FALSE;

RETURN TRUE END;

RETURN FALSE END ParseBoolean;

VAR i,n,d:CARDINAL;

RETURN TRUE END ParseCardinal;

IMPLEMENTATION MODULE M2Math;

PROCEDURE GCD(a,b:CARDINAL):CARDINAL;

VAR t:CARDINAL;

BEGIN WHILE b#0 DO t:=a MOD b;

a:=b;

b:=t END;

RETURN a END GCD;

PROCEDURE LCM(a,b:CARDINAL;

VAR result:CARDINAL):BOOLEAN;

VAR g:CARDINAL;

BEGIN IF(a=0)OR(b=0)THEN result:=0;

RETURN TRUE END;

g:=GCD(a,b);

a:=a DIV g;

IF a>MAX(CARDINAL)DIV b THEN RETURN FALSE END;

result:=a*b;

RETURN TRUE END LCM;

PROCEDURE IsEven(n:CARDINAL):BOOLEAN;

BEGIN RETURN n MOD 2=0 END IsEven;

PROCEDURE IsOdd(n:CARDINAL):BOOLEAN;

BEGIN RETURN n MOD 2=1 END IsOdd;

PROCEDURE MinCard(a,b:CARDINAL):CARDINAL;

BEGIN IF a<b THEN RETURN a ELSE RETURN b END END MinCard;

PROCEDURE MaxCard(a,b:CARDINAL):CARDINAL;

BEGIN IF a>b THEN RETURN a ELSE RETURN b END END MaxCard;

PROCEDURE Clamp(n,lo,hi:CARDINAL):CARDINAL;

BEGIN IF lo>hi THEN RETURN lo END;

IF n<lo THEN RETURN lo END;

IF n>hi THEN RETURN hi END;

RETURN n END Clamp;

VAR result:CARDINAL):BOOLEAN;

VAR i:CARDINAL;

BEGIN result:=1;

IMPLEMENTATION MODULE M2NetFrame;

PROCEDURE Encode(payload:ARRAY OF CHAR;

VAR frame:ARRAY OF CHAR;

VAR frameLen:CARDINAL):BOOLEAN;

VAR n,i:CARDINAL;

BEGIN n:=0;

WHILE(n<=HIGH(payload))AND(payload[n]#0C)DO INC(n)END;

IF n+4>HIGH(frame)THEN RETURN FALSE END;

frame[0]:=CHR(n DIV 16777216);

frame[1]:=CHR((n DIV 65536)MOD 256);

frame[2]:=CHR((n DIV 256)MOD 256);

frame[3]:=CHR(n MOD 256);

FOR i:=0 TO n-1 DO frame[i+4]:=payload[i]END;

frame[n+4]:=0C;

frameLen:=n+4;

RETURN TRUE END Encode;

PROCEDURE Decode(frame:ARRAY OF CHAR;

frameLen:CARDINAL;

VAR payload:ARRAY OF CHAR;

VAR payloadLen:CARDINAL):BOOLEAN;

VAR n,i:CARDINAL;

BEGIN IF frameLen<4 THEN RETURN FALSE END;

n:=ORD(frame[0])*16777216+ORD(frame[1])*65536+ORD(frame[2])*256+ORD(frame[3]);

IF n+4#frameLen THEN RETURN FALSE END;

IF n>HIGH(payload)THEN RETURN FALSE END;

FOR i:=0 TO n-1 DO payload[i]:=frame[i+4]END;

payload[n]:=0C;

payloadLen:=n;

RETURN TRUE END Decode;

END M2NetFrame.

BEGIN c:=crc;

PROCEDURE IsSpace(c:CHAR):BOOLEAN;

PROCEDURE ToLower(c:CHAR):CHAR;

END M2Chars.

########## src/text/M2Chars.mod ##########

IMPLEMENTATION MODULE M2Chars;

PROCEDURE IsAlpha(c:CHAR):BOOLEAN;

BEGIN RETURN((c>='A')AND(c<='Z'))OR((c>='a')AND(c<='z'))END IsAlpha;

PROCEDURE IsDigit(c:CHAR):BOOLEAN;

BEGIN RETURN(c>='0')AND(c<='9')END IsDigit;

PROCEDURE IsHexDigit(c:CHAR):BOOLEAN;

BEGIN RETURN IsDigit(c)OR((c>='A')AND(c<='F'))OR((c>='a')AND(c<='f'))END IsHexDigit;

PROCEDURE IsAlNum(c:CHAR):BOOLEAN;

BEGIN RETURN IsAlpha(c)OR IsDigit(c)END IsAlNum;

PROCEDURE IsSpace(c:CHAR):BOOLEAN;

BEGIN RETURN(c=' ')OR(c=9C)OR(c=10C)OR(c=13C)END IsSpace;

PROCEDURE ToUpper(c:CHAR):CHAR;

BEGIN IF(c>='a')AND(c<='z')THEN RETURN CHR(ORD(c)-32)ELSE RETURN c END END ToUpper;

PROCEDURE ToLower(c:CHAR):CHAR;

BEGIN IF(c>='A')AND(c<='Z')THEN RETURN CHR(ORD(c)+32)ELSE RETURN c END END ToLower;

PROCEDURE HexValue(c:CHAR):CARDINAL;

BEGIN IF IsDigit(c)THEN RETURN ORD(c)-ORD('0')END;

IF(c>='A')AND(c<='F')THEN RETURN ORD(c)-ORD('A')+10 END;

IF(c>='a')AND(c<='f')THEN RETURN ORD(c)-ORD('a')+10 END;

RETURN MAX(CARDINAL)END HexValue;

END M2Chars.

########## src/text/M2Strings.mod ##########

IMPLEMENTATION MODULE M2Strings;

PROCEDURE Length(s:ARRAY OF CHAR):CARDINAL;

VAR i:CARDINAL;

BEGIN i:=0;

WHILE(i<=HIGH(s))AND(s[i]#0C)DO INC(i)END;

RETURN i END Length;

PROCEDURE Copy(src:ARRAY OF CHAR;

VAR dst:ARRAY OF CHAR);

VAR i:CARDINAL;

BEGIN i:=0;

WHILE(i<Length(src))AND(i<HIGH(dst))DO dst[i]:=src[i];

INC(i)END;

IF i<=HIGH(dst)THEN dst[i]:=0C END END Copy;

PROCEDURE Append(src:ARRAY OF CHAR;

VAR dst:ARRAY OF CHAR);

VAR i,j:CARDINAL;

BEGIN i:=Length(dst);

j:=0;

WHILE(j<Length(src))AND(i<HIGH(dst))DO dst[i]:=src[j];

INC(i);

INC(j)END;

IF i<=HIGH(dst)THEN dst[i]:=0C END END Append;

PROCEDURE Compare(a,b:ARRAY OF CHAR):INTEGER;

VAR i:CARDINAL;

BEGIN i:=0;

LOOP IF a[i]<b[i]THEN RETURN -1 END;

IF a[i]>b[i]THEN RETURN 1 END;

IF(a[i]=0C)OR(i>=HIGH(a))OR(i>=HIGH(b))THEN EXIT END;

INC(i)END;

RETURN 0 END Compare;

PROCEDURE Equal(a,b:ARRAY OF CHAR):BOOLEAN;

BEGIN RETURN Compare(a,b)=0 END Equal;

PROCEDURE Index(s,pattern:ARRAY OF CHAR):INTEGER;

VAR i,j,ns,np:CARDINAL;

BEGIN ns:=Length(s);

np:=Length(pattern);

IF np=0 THEN RETURN 0 END;

IF np>ns THEN RETURN -1 END;

i:=0;

WHILE i+np<=ns DO j:=0;

WHILE(j<np)AND(s[i+j]=pattern[j])DO INC(j)END;

IF j=np THEN RETURN VAL(INTEGER,i)END;

INC(i)END;

RETURN -1 END Index;

PROCEDURE RIndex(s,pattern:ARRAY OF CHAR):INTEGER;

VAR i,j,ns,np:CARDINAL;

BEGIN ns:=Length(s);

np:=Length(pattern);

IF np=0 THEN RETURN VAL(INTEGER,ns)END;

IF np>ns THEN RETURN -1 END;

i:=ns-np+1;

WHILE i>0 DO DEC(i);

j:=0;

WHILE(j<np)AND(s[i+j]=pattern[j])DO INC(j)END;

IF j=np THEN RETURN VAL(INTEGER,i)END END;

RETURN -1 END RIndex;

PROCEDURE SubString(s:ARRAY OF CHAR;

start,count:CARDINAL;

VAR dst:ARRAY OF CHAR);

VAR i:CARDINAL;

BEGIN i:=0;

WHILE(i<count)AND(start+i<Length(s))AND(i<HIGH(dst))DO dst[i]:=s[start+i];

INC(i)END;

IF i<=HIGH(dst)THEN dst[i]:=0C END END SubString;

PROCEDURE Trim(s:ARRAY OF CHAR;

VAR dst:ARRAY OF CHAR);

VAR i,n:CARDINAL;

BEGIN n:=Length(s);

i:=0;

WHILE(i<n)AND((s[i]=' ')OR(s[i]=9C)OR(s[i]=10C)OR(s[i]=13C))DO INC(i)END;

WHILE n>i DO IF(s[n-1]=' ')OR(s[n-1]=9C)OR(s[n-1]=10C)OR(s[n-1]=13C)THEN DEC(n)ELSE EXIT END END;

SubString(s,i,n-i,dst)END Trim;

PROCEDURE ToUpper(s:ARRAY OF CHAR;

VAR dst:ARRAY OF CHAR);

VAR i:CARDINAL;

c:CHAR;

BEGIN i:=0;

WHILE(i<Length(s))AND(i<HIGH(dst))DO c:=s[i];

IF(c>='a')AND(c<='z')THEN c:=CHR(ORD(c)-32)END;

dst[i]:=c;

INC(i)END;

IF i<=HIGH(dst)THEN dst[i]:=0C END END ToUpper;

PROCEDURE ToLower(s:ARRAY OF CHAR;

VAR dst:ARRAY OF CHAR);

VAR i:CARDINAL;

c:CHAR;

BEGIN i:=0;

WHILE(i<Length(s))AND(i<HIGH(dst))DO c:=s[i];

IF(c>='A')AND(c<='Z')THEN c:=CHR(ORD(c)+32)END;

dst[i]:=c;

INC(i)END;

IF i<=HIGH(dst)THEN dst[i]:=0C END END ToLower;

PROCEDURE StartsWith(s,prefix:ARRAY OF CHAR):BOOLEAN;

VAR i:CARDINAL;

BEGIN i:=0;

WHILE i<Length(prefix)DO IF(i>=Length(s))OR(s[i]#prefix[i])THEN RETURN FALSE END;

INC(i)END;

RETURN TRUE END StartsWith;

PROCEDURE EndsWith(s,suffix:ARRAY OF CHAR):BOOLEAN;

VAR i,ns,np:CARDINAL;

BEGIN ns:=Length(s);

np:=Length(suffix);

IF np>ns THEN RETURN FALSE END;

i:=0;

WHILE i<np DO IF s[ns-np+i]#suffix[i]THEN RETURN FALSE END;

INC(i)END;

RETURN TRUE END EndsWith;

PROCEDURE IsEmpty(s:ARRAY OF CHAR):BOOLEAN;

BEGIN RETURN Length(s)=0 END IsEmpty;

PROCEDURE Fill(ch:CHAR;

n:CARDINAL;

VAR dst:ARRAY OF CHAR);

VAR i:CARDINAL;

BEGIN i:=0;

WHILE(i<n)AND(i<HIGH(dst))DO dst[i]:=ch;

INC(i)END;

IF i<=HIGH(dst)THEN dst[i]:=0C END END Fill;

PROCEDURE Reverse(s:ARRAY OF CHAR;

VAR dst:ARRAY OF CHAR);

VAR i,n:CARDINAL;

BEGIN n:=Length(s);

i:=0;

WHILE(i<n)AND(i<HIGH(dst))DO dst[i]:=s[n-i-1];

INC(i)END;

IF i<=HIGH(dst)THEN dst[i]:=0C END END Reverse;

END M2Strings.

########## src/text/M2UTF8.mod ##########

IMPLEMENTATION MODULE M2UTF8;

PROCEDURE DecodeRune(s:ARRAY OF CHAR;

pos:CARDINAL;

VAR rune,width:CARDINAL):BOOLEAN;

VAR n,b,b1,b2,b3:CARDINAL;

BEGIN n:=0;

WHILE(n<=HIGH(s))AND(s[n]#0C)DO INC(n)END;

IF pos>=n THEN RETURN FALSE END;

b:=ORD(s[pos]);

IF b<128 THEN rune:=b;

width:=1;

RETURN TRUE END;

IF(b>=194)AND(b<=223)THEN width:=2 ELSIF(b>=224)AND(b<=239)THEN width:=3 ELSIF(b>=240)AND(b<=244)THEN width:=4 ELSE RETURN FALSE END;

IF pos+width>n THEN RETURN FALSE END;

b1:=ORD(s[pos+1]);

IF(b1<128)OR(b1>191)THEN RETURN FALSE END;

IF width=2 THEN rune:=(b MOD 32)*64+(b1 MOD 64);

RETURN TRUE END;

b2:=ORD(s[pos+2]);

IF(b2<128)OR(b2>191)THEN RETURN FALSE END;

IF width=3 THEN IF(b=224)AND(b1<160)THEN RETURN FALSE END;

IF(b=237)AND(b1>159)THEN RETURN FALSE END;

rune:=(b MOD 16)*4096+(b1 MOD 64)*64+(b2 MOD 64);

RETURN TRUE END;

b3:=ORD(s[pos+3]);

IF(b3<128)OR(b3>191)THEN RETURN FALSE END;

IF(b=240)AND(b1<144)THEN RETURN FALSE END;

IF(b=244)AND(b1>143)THEN RETURN FALSE END;

rune:=(b MOD 8)*262144+(b1 MOD 64)*4096+(b2 MOD 64)*64+(b3 MOD 64);

RETURN TRUE END DecodeRune;

PROCEDURE IsValid(s:ARRAY OF CHAR):BOOLEAN;

VAR p,r,w,n:CARDINAL;

BEGIN n:=0;

WHILE(n<=HIGH(s))AND(s[n]#0C)DO INC(n)END;

p:=0;

WHILE p<n DO IF NOT DecodeRune(s,p,r,w)THEN RETURN FALSE END;

INC(p,w)END;

RETURN TRUE END IsValid;

PROCEDURE CountRunes(s:ARRAY OF CHAR;

VAR count:CARDINAL):BOOLEAN;

VAR p,r,w,n:CARDINAL;

BEGIN n:=0;

WHILE(n<=HIGH(s))AND(s[n]#0C)DO INC(n)END;

p:=0;

count:=0;

WHILE p<n DO IF NOT DecodeRune(s,p,r,w)THEN RETURN FALSE END;

INC(count);

INC(p,w)END;

RETURN TRUE END CountRunes;

END M2UTF8.

END M2Time.

########## src/time/M2Time.mod ##########

IMPLEMENTATION MODULE M2Time;

PROCEDURE IsLeapYear(year:CARDINAL):BOOLEAN;

BEGIN RETURN(year MOD 4=0)AND((year MOD 100#0)OR(year MOD 400=0))END IsLeapYear;

PROCEDURE DaysInMonth(year,month:CARDINAL):CARDINAL;

BEGIN CASE month OF 1,3,5,7,8,10,12:RETURN 31|4,6,9,11:RETURN 30|2:IF IsLeapYear(year)THEN RETURN 29 ELSE RETURN 28 END ELSE RETURN 0 END END DaysInMonth;

PROCEDURE IsValidDate(d:Date):BOOLEAN;

BEGIN RETURN(d.year>0)AND(d.month>=1)AND(d.month<=12)AND(d.day>=1)AND(d.day<=DaysInMonth(d.year,d.month))END IsValidDate;

PROCEDURE DayOfYear(d:Date):CARDINAL;

VAR m,n:CARDINAL;

BEGIN IF NOT IsValidDate(d)THEN RETURN 0 END;

n:=d.day;

FOR m:=1 TO d.month-1 DO INC(n,DaysInMonth(d.year,m))END;

RETURN n END DayOfYear;

END M2Time.
